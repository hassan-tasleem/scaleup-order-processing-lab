### Maven Parent overrides

Due to Maven's design, elements are inherited from the parent POM to the project POM.
While most of the inheritance is fine, it also inherits unwanted elements like `<license>` and `<developers>` from the parent.
To prevent this, the project POM contains empty overrides for these elements.
If you manually switch to a different parent and actually want the inheritance, you need to remove those overrides.

# Apache Maven Wrapper startup batch script, version 3.3.4
#
# Optional ENV vars
# -----------------
#   JAVA_HOME - location of a JDK home dir, required when download maven via java source
#   MVNW_REPOURL - repo url base for downloading maven distribution
#   MVNW_USERNAME/MVNW_PASSWORD - user and password for downloading maven
#   MVNW_VERBOSE - true: enable verbose log; debug: trace the mvnw script; others: silence the output
# ----------------------------------------------------------------------------

set -euf
[ "${MVNW_VERBOSE-}" != debug ] || set -x

# OS specific support.
native_path() { printf %s\\n "$1"; }
case "$(uname)" in
CYGWIN* | MINGW*)
  [ -z "${JAVA_HOME-}" ] || JAVA_HOME="$(cygpath --unix "$JAVA_HOME")"
  native_path() { cygpath --path --windows "$1"; }
  ;;
esac

# set JAVACMD and JAVACCMD
set_java_home() {
  # For Cygwin and MinGW, ensure paths are in Unix format before anything is touched
  if [ -n "${JAVA_HOME-}" ]; then
    if [ -x "$JAVA_HOME/jre/sh/java" ]; then
      # IBM's JDK on AIX uses strange locations for the executables
      JAVACMD="$JAVA_HOME/jre/sh/java"
      JAVACCMD="$JAVA_HOME/jre/sh/javac"
    else
      JAVACMD="$JAVA_HOME/bin/java"
      JAVACCMD="$JAVA_HOME/bin/javac"

      if [ ! -x "$JAVACMD" ] || [ ! -x "$JAVACCMD" ]; then
        echo "The JAVA_HOME environment variable is not defined correctly, so mvnw cannot run." >&2
        echo "JAVA_HOME is set to \"$JAVA_HOME\", but \"\$JAVA_HOME/bin/java\" or \"\$JAVA_HOME/bin/javac\" does not exist." >&2
        return 1
      fi
    fi
  else
    JAVACMD="$(
      'set' +e
      'unset' -f command 2>/dev/null
      'command' -v java
    )" || :
    JAVACCMD="$(
      'set' +e
      'unset' -f command 2>/dev/null
      'command' -v javac
    )" || :

    if [ ! -x "${JAVACMD-}" ] || [ ! -x "${JAVACCMD-}" ]; then
      echo "The java/javac command does not exist in PATH nor is JAVA_HOME set, so mvnw cannot run." >&2
      return 1
    fi
  fi
}

# hash string like Java String::hashCode
hash_string() {
  str="${1:-}" h=0
  while [ -n "$str" ]; do
    char="${str%"${str#?}"}"
    h=$(((h * 31 + $(LC_CTYPE=C printf %d "'$char")) % 4294967296))
    str="${str#?}"
  done
  printf %x\\n $h
}

verbose() { :; }
[ "${MVNW_VERBOSE-}" != true ] || verbose() { printf %s\\n "${1-}"; }

die() {
  printf %s\\n "$1" >&2
  exit 1
}

trim() {
  # MWRAPPER-139:
  #   Trims trailing and leading whitespace, carriage returns, tabs, and linefeeds.
  #   Needed for removing poorly interpreted newline sequences when running in more
  #   exotic environments such as mingw bash on Windows.
  printf "%s" "${1}" | tr -d '[:space:]'
}

scriptDir="$(dirname "$0")"
scriptName="$(basename "$0")"

# parse distributionUrl and optional distributionSha256Sum, requires .mvn/wrapper/maven-wrapper.properties
while IFS="=" read -r key value; do
  case "${key-}" in
  distributionUrl) distributionUrl=$(trim "${value-}") ;;
  distributionSha256Sum) distributionSha256Sum=$(trim "${value-}") ;;
  esac
done <"$scriptDir/.mvn/wrapper/maven-wrapper.properties"
[ -n "${distributionUrl-}" ] || die "cannot read distributionUrl property in $scriptDir/.mvn/wrapper/maven-wrapper.properties"

case "${distributionUrl##*/}" in
maven-mvnd-*bin.*)
  MVN_CMD=mvnd.sh _MVNW_REPO_PATTERN=/maven/mvnd/
  case "${PROCESSOR_ARCHITECTURE-}${PROCESSOR_ARCHITEW6432-}:$(uname -a)" in
  *AMD64:CYGWIN* | *AMD64:MINGW*) distributionPlatform=windows-amd64 ;;
  :Darwin*x86_64) distributionPlatform=darwin-amd64 ;;
  :Darwin*arm64) distributionPlatform=darwin-aarch64 ;;
  :Linux*x86_64*) distributionPlatform=linux-amd64 ;;
  *)
    echo "Cannot detect native platform for mvnd on $(uname)-$(uname -m), use pure java version" >&2
    distributionPlatform=linux-amd64
    ;;
  esac
  distributionUrl="${distributionUrl%-bin.*}-$distributionPlatform.zip"
  ;;
maven-mvnd-*) MVN_CMD=mvnd.sh _MVNW_REPO_PATTERN=/maven/mvnd/ ;;
*) MVN_CMD="mvn${scriptName#mvnw}" _MVNW_REPO_PATTERN=/org/apache/maven/ ;;
esac

# apply MVNW_REPOURL and calculate MAVEN_HOME
# maven home pattern: ~/.m2/wrapper/dists/{apache-maven-<version>,maven-mvnd-<version>-<platform>}/<hash>
[ -z "${MVNW_REPOURL-}" ] || distributionUrl="$MVNW_REPOURL$_MVNW_REPO_PATTERN${distributionUrl#*"$_MVNW_REPO_PATTERN"}"
distributionUrlName="${distributionUrl##*/}"
distributionUrlNameMain="${distributionUrlName%.*}"
distributionUrlNameMain="${distributionUrlNameMain%-bin}"
MAVEN_USER_HOME="${MAVEN_USER_HOME:-${HOME}/.m2}"
MAVEN_HOME="${MAVEN_USER_HOME}/wrapper/dists/${distributionUrlNameMain-}/$(hash_string "$distributionUrl")"

exec_maven() {
  unset MVNW_VERBOSE MVNW_USERNAME MVNW_PASSWORD MVNW_REPOURL || :
  exec "$MAVEN_HOME/bin/$MVN_CMD" "$@" || die "cannot exec $MAVEN_HOME/bin/$MVN_CMD"
}

if [ -d "$MAVEN_HOME" ]; then
  verbose "found existing MAVEN_HOME at $MAVEN_HOME"
  exec_maven "$@"
fi

case "${distributionUrl-}" in
*?-bin.zip | *?maven-mvnd-?*-?*.zip) ;;
*) die "distributionUrl is not valid, must match *-bin.zip or maven-mvnd-*.zip, but found '${distributionUrl-}'" ;;
esac

# prepare tmp dir
if TMP_DOWNLOAD_DIR="$(mktemp -d)" && [ -d "$TMP_DOWNLOAD_DIR" ]; then
  clean() { rm -rf -- "$TMP_DOWNLOAD_DIR"; }
  trap clean HUP INT TERM EXIT
else
  die "cannot create temp dir"
fi

mkdir -p -- "${MAVEN_HOME%/*}"

# Download and Install Apache Maven
verbose "Couldn't find MAVEN_HOME, downloading and installing it ..."
verbose "Downloading from: $distributionUrl"
verbose "Downloading to: $TMP_DOWNLOAD_DIR/$distributionUrlName"

# select .zip or .tar.gz
if ! command -v unzip >/dev/null; then
  distributionUrl="${distributionUrl%.zip}.tar.gz"
  distributionUrlName="${distributionUrl##*/}"
fi

# verbose opt
__MVNW_QUIET_WGET=--quiet __MVNW_QUIET_CURL=--silent __MVNW_QUIET_UNZIP=-q __MVNW_QUIET_TAR=''
[ "${MVNW_VERBOSE-}" != true ] || __MVNW_QUIET_WGET='' __MVNW_QUIET_CURL='' __MVNW_QUIET_UNZIP='' __MVNW_QUIET_TAR=v

# normalize http auth
case "${MVNW_PASSWORD:+has-password}" in
'') MVNW_USERNAME='' MVNW_PASSWORD='' ;;
has-password) [ -n "${MVNW_USERNAME-}" ] || MVNW_USERNAME='' MVNW_PASSWORD='' ;;
esac

if [ -z "${MVNW_USERNAME-}" ] && command -v wget >/dev/null; then
  verbose "Found wget ... using wget"
  wget ${__MVNW_QUIET_WGET:+"$__MVNW_QUIET_WGET"} "$distributionUrl" -O "$TMP_DOWNLOAD_DIR/$distributionUrlName" || die "wget: Failed to fetch $distributionUrl"
elif [ -z "${MVNW_USERNAME-}" ] && command -v curl >/dev/null; then
  verbose "Found curl ... using curl"
  curl ${__MVNW_QUIET_CURL:+"$__MVNW_QUIET_CURL"} -f -L -o "$TMP_DOWNLOAD_DIR/$distributionUrlName" "$distributionUrl" || die "curl: Failed to fetch $distributionUrl"
elif set_java_home; then
  verbose "Falling back to use Java to download"
  javaSource="$TMP_DOWNLOAD_DIR/Downloader.java"
  targetZip="$TMP_DOWNLOAD_DIR/$distributionUrlName"
  cat >"$javaSource" <<-END
	public class Downloader extends java.net.Authenticator
	{
	  protected java.net.PasswordAuthentication getPasswordAuthentication()
	  {
	    return new java.net.PasswordAuthentication( System.getenv( "MVNW_USERNAME" ), System.getenv( "MVNW_PASSWORD" ).toCharArray() );
	  }
	  public static void main( String[] args ) throws Exception
	  {
	    setDefault( new Downloader() );
	    java.nio.file.Files.copy( java.net.URI.create( args[0] ).toURL().openStream(), java.nio.file.Paths.get( args[1] ).toAbsolutePath().normalize() );
	  }
	}
	END
  # For Cygwin/MinGW, switch paths to Windows format before running javac and java
  verbose " - Compiling Downloader.java ..."
  "$(native_path "$JAVACCMD")" "$(native_path "$javaSource")" || die "Failed to compile Downloader.java"
  verbose " - Running Downloader.java ..."
  "$(native_path "$JAVACMD")" -cp "$(native_path "$TMP_DOWNLOAD_DIR")" Downloader "$distributionUrl" "$(native_path "$targetZip")"
fi

# If specified, validate the SHA-256 sum of the Maven distribution zip file
if [ -n "${distributionSha256Sum-}" ]; then
  distributionSha256Result=false
  if [ "$MVN_CMD" = mvnd.sh ]; then
    echo "Checksum validation is not supported for maven-mvnd." >&2
    echo "Please disable validation by removing 'distributionSha256Sum' from your maven-wrapper.properties." >&2
    exit 1
  elif command -v sha256sum >/dev/null; then
    if echo "$distributionSha256Sum  $TMP_DOWNLOAD_DIR/$distributionUrlName" | sha256sum -c - >/dev/null 2>&1; then
      distributionSha256Result=true
    fi
  elif command -v shasum >/dev/null; then
    if echo "$distributionSha256Sum  $TMP_DOWNLOAD_DIR/$distributionUrlName" | shasum -a 256 -c >/dev/null 2>&1; then
      distributionSha256Result=true
    fi
  else
    echo "Checksum validation was requested but neither 'sha256sum' or 'shasum' are available." >&2
    echo "Please install either command, or disable validation by removing 'distributionSha256Sum' from your maven-wrapper.properties." >&2
    exit 1
  fi
  if [ $distributionSha256Result = false ]; then
    echo "Error: Failed to validate Maven distribution SHA-256, your Maven distribution might be compromised." >&2
    echo "If you updated your Maven version, you need to update the specified distributionSha256Sum property." >&2
    exit 1
  fi
fi

# unzip and move
if command -v unzip >/dev/null; then
  unzip ${__MVNW_QUIET_UNZIP:+"$__MVNW_QUIET_UNZIP"} "$TMP_DOWNLOAD_DIR/$distributionUrlName" -d "$TMP_DOWNLOAD_DIR" || die "failed to unzip"
else

import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;

import org.springframework.stereotype.Service;

    private final SimulationSettings simulationSettings;

    public ResetService(JdbcTemplate jdbcTemplate, SimulationSettings simulationSettings) {
        this.jdbcTemplate = jdbcTemplate;
        this.simulationSettings = simulationSettings;
    }

    @Transactional
    public ResetResponse reset(SeedProfile profile) {
        // Empty everything except Flyway's own history table. Ids start from 1 again.
        jdbcTemplate.execute("""
                TRUNCATE TABLE order_items, payments, notifications, orders,
                               inventory, products, customers
                RESTART IDENTITY CASCADE
                """);

        jdbcTemplate.update("""
                INSERT INTO customers (name, email)
                SELECT 'Customer ' || n, 'customer_' || n || '@example.test'
                FROM generate_series(1, ?) AS n
                ORDER BY n
                """, profile.getCustomers());

        jdbcTemplate.update("""
                INSERT INTO products (sku, name, category, price)
                SELECT CASE
                           WHEN n <= 20 THEN 'HOT-' || lpad(n::text, 3, '0')
                           WHEN n = 21  THEN 'FLASH-001'
                           WHEN n = 22  THEN 'ROW-HOT-001'
                           ELSE 'SKU-' || lpad(n::text, 6, '0')
                       END,
                       'Product ' || n,
                       CASE
                           WHEN n <= 20 THEN 'hot'
                           WHEN n = 21  THEN 'flash'
                           WHEN n = 22  THEN 'hot-row'
                           ELSE 'category-' || (n % 10)
                       END,
                       (5 + ((n * 37) % 1000) / 10.0)::numeric(12,2)
                FROM generate_series(1, ?) AS n
                ORDER BY n
                """, profile.getProducts());

        jdbcTemplate.update("""
                INSERT INTO inventory (product_id, initial_qty, available_qty, reserved_qty)
                SELECT id,
                       CASE WHEN sku = 'FLASH-001' THEN 10 ELSE 1000000 END,
                       CASE WHEN sku = 'FLASH-001' THEN 10 ELSE 1000000 END,
                       0
                FROM products
                ORDER BY id
                """);

        // Refresh PostgreSQL's statistics so query plans match the new data.
        jdbcTemplate.execute("""
                ANALYZE customers, products, inventory, orders, order_items, payments, notifications
                """);

        // Every run starts without injected delays or failures.
        simulationSettings.reset();

        return new ResetResponse(
                profile.name(),
                count("customers"),
                count("products"),
                count("inventory"));
    }

    private long count(String table) {
        Long value = jdbcTemplate.queryForObject("SELECT count(*) FROM " + table, Long.class);
        return value == null ? 0 : value;
    }
}

package com.scaleup.admin;

import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;

@Service
@ConditionalOnProperty(name = "scaleup.admin.enabled", havingValue = "true")
public class VerifyService {

    private static final List<String> TABLES = List.of(
            "customers", "products", "inventory", "orders",
            "order_items", "payments", "notifications");

    private final JdbcTemplate jdbcTemplate;

    public VerifyService(JdbcTemplate jdbcTemplate) {
        this.jdbcTemplate = jdbcTemplate;
    }

    @Transactional(readOnly = true)
    public VerifyResponse verify() {
        Map<String, Long> rowCounts = new LinkedHashMap<>();
        for (String table : TABLES) {
            rowCounts.put(table, queryForLong("SELECT count(*) FROM " + table));
        }

        Map<String, Long> ordersByStatus = new LinkedHashMap<>();
        List<Map<String, Object>> statusRows = jdbcTemplate.queryForList(
                "SELECT status, count(*) AS total FROM orders GROUP BY status ORDER BY status");
        for (Map<String, Object> row : statusRows) {
            ordersByStatus.put((String) row.get("status"), ((Number) row.get("total")).longValue());
        }
        long inFlight = ordersByStatus.getOrDefault("PENDING", 0L)
                + ordersByStatus.getOrDefault("STOCK_RESERVED", 0L)
                + ordersByStatus.getOrDefault("PAID", 0L);

        List<VerifyResponse.Check> checks = new ArrayList<>();

        checks.add(check("noNegativeStock",
                "SELECT count(*) FROM inventory WHERE available_qty < 0 OR reserved_qty < 0"));

        checks.add(check("stockBalances", """
                SELECT count(*)
                FROM inventory i
                LEFT JOIN (SELECT oi.product_id, sum(oi.quantity) AS sold
                           FROM order_items oi
                           JOIN orders o ON o.id = oi.order_id
                           WHERE o.status IN ('PAID', 'CONFIRMED')
                           GROUP BY oi.product_id) s ON s.product_id = i.product_id
                WHERE i.initial_qty <> i.available_qty + i.reserved_qty + coalesce(s.sold, 0)
                """));

        checks.add(check("everyProductHasInventory", """
                SELECT count(*)
                FROM products p
                LEFT JOIN inventory i ON i.product_id = p.id
                WHERE i.product_id IS NULL
                """));

        checks.add(check("ordersHaveItems", """
                SELECT count(*)
                FROM orders o
                WHERE NOT EXISTS (SELECT 1 FROM order_items oi WHERE oi.order_id = o.id)
                """));

        checks.add(check("orderTotalsMatchItems", """
                SELECT count(*)
                FROM orders o
                JOIN (SELECT order_id, sum(quantity * unit_price) AS total
                      FROM order_items GROUP BY order_id) t ON t.order_id = o.id
                WHERE o.total_amount <> t.total
                """));

        checks.add(check("specialProductsSeeded", """
                SELECT 2 - count(*)
                FROM products p
                JOIN inventory i ON i.product_id = p.id
                WHERE (p.sku = 'FLASH-001' AND i.initial_qty = 10)
                   OR (p.sku = 'ROW-HOT-001' AND i.initial_qty = 1000000)
                """));

        boolean allPassed = checks.stream().allMatch(VerifyResponse.Check::passed);

        return new VerifyResponse(allPassed, seedChecksum(), inFlight,
                rowCounts, ordersByStatus, checks);
    }

    private VerifyResponse.Check check(String name, String sql) {
        long violations = queryForLong(sql);
        return new VerifyResponse.Check(name, violations == 0, violations);
    }

    private String seedChecksum() {
        return jdbcTemplate.queryForObject("""
                SELECT md5(
                    coalesce((SELECT string_agg(id || ':' || name || ':' || email, ',' ORDER BY id)
                              FROM customers), '') || '|' ||
                    coalesce((SELECT string_agg(id || ':' || sku || ':' || name || ':' || category || ':' || price,
                                                ',' ORDER BY id)
                              FROM products), '') || '|' ||
                    coalesce((SELECT string_agg(product_id || ':' || initial_qty, ',' ORDER BY product_id)
                              FROM inventory), '')
                )
                """, String.class);
    }

    private long queryForLong(String sql) {
        Long value = jdbcTemplate.queryForObject(sql, Long.class);
        return value == null ? 0 : value;
    }
}

    }

    @ExceptionHandler(PaymentTimeoutException.class)
    public ResponseEntity<ApiError> handlePaymentTimeout(PaymentTimeoutException ex) {
        ApiError body = new ApiError(504, "Gateway Timeout", ex.getMessage(), Instant.now());
        return ResponseEntity.status(HttpStatus.GATEWAY_TIMEOUT).body(body);
    }

    // Baseline behaviour (P07): a failed notification fails the request, although the
    // payment already succeeded and the order stays PAID.
    @ExceptionHandler(NotificationFailedException.class)
    public ResponseEntity<ApiError> handleNotificationFailed(NotificationFailedException ex) {
        ApiError body = new ApiError(500, "Internal Server Error", ex.getMessage(), Instant.now());
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(body);
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiError> handleValidation(MethodArgumentNotValidException ex) {
        String message = ex.getBindingResult().getFieldErrors().stream()
                .map(error -> error.getField() + " " + error.getDefaultMessage())
                .collect(Collectors.joining("; "));
        ApiError body = new ApiError(400, "Bad Request", message, Instant.now());
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }

    @ExceptionHandler(HttpMessageNotReadableException.class)
    public ResponseEntity<ApiError> handleUnreadableBody(HttpMessageNotReadableException ex) {
        ApiError body = new ApiError(400, "Bad Request", "Malformed request body", Instant.now());
        return ResponseEntity.status(HttpStatus.BAD_REQUEST).body(body);
    }
}

package com.scaleup.inventory;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Modifying;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

public interface InventoryRepository extends JpaRepository<Inventory, Long> {

    // Moves stock from available to reserved, only if enough is available.
    // Returns the number of rows changed: 1 = reserved, 0 = not enough stock.
    @Modifying
    @Query(value = """
            UPDATE inventory
            SET available_qty = available_qty - :quantity,
                reserved_qty  = reserved_qty + :quantity,
                version       = version + 1,
                updated_at    = now()
            WHERE product_id = :productId
              AND available_qty >= :quantity
            """, nativeQuery = true)
    int reserve(@Param("productId") Long productId, @Param("quantity") int quantity);

    // Moves stock back from reserved to available, only if that much is reserved.
    // Returns the number of rows changed: 1 = released, 0 = fewer units were reserved.
    @Modifying
    @Query(value = """
            UPDATE inventory
            SET available_qty = available_qty + :quantity,
                reserved_qty  = reserved_qty - :quantity,
                version       = version + 1,
                updated_at    = now()
            WHERE product_id = :productId
              AND reserved_qty >= :quantity
            """, nativeQuery = true)
    int release(@Param("productId") Long productId, @Param("quantity") int quantity);

    // Marks reserved units as sold: they leave the reserved count for good.
    // Returns the number of rows changed: 1 = marked, 0 = fewer units were reserved.
    @Modifying
    @Query(value = """
            UPDATE inventory
            SET reserved_qty = reserved_qty - :quantity,
                version      = version + 1,
                updated_at   = now()
            WHERE product_id = :productId
              AND reserved_qty >= :quantity
            """, nativeQuery = true)
    int markSold(@Param("productId") Long productId, @Param("quantity") int quantity);
}

package com.scaleup.inventory;

import com.scaleup.common.InsufficientStockException;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

@Service
public class InventoryService {

    private final InventoryRepository inventoryRepository;

    public InventoryService(InventoryRepository inventoryRepository) {
        this.inventoryRepository = inventoryRepository;
    }

    @Transactional(readOnly = true)
    public int getAvailableQuantity(Long productId) {
        return inventoryRepository.findById(productId)
                .map(Inventory::getAvailableQty)
                .orElse(0);
    }

    // Joins the caller's transaction: if a later step of the order fails,
    // the reservation is undone together with everything else in that transaction.
    @Transactional
    public void reserve(Long productId, int quantity) {
        int changedRows = inventoryRepository.reserve(productId, quantity);
        if (changedRows == 0) {
            throw new InsufficientStockException(productId, quantity);
        }
    }

    @Transactional
    public void release(Long productId, int quantity) {
        int changedRows = inventoryRepository.release(productId, quantity);
        if (changedRows == 0) {
            throw new IllegalStateException(
                    "Cannot release " + quantity + " units of product " + productId
                            + ": fewer are reserved");
        }
    }

    @Transactional
    public void markSold(Long productId, int quantity) {
        int changedRows = inventoryRepository.markSold(productId, quantity);
        if (changedRows == 0) {
            throw new IllegalStateException(
                    "Cannot mark " + quantity + " units of product " + productId
                            + " as sold: fewer are reserved");
        }
    }
}

package com.scaleup.order;

import com.scaleup.common.BadRequestException;
import com.scaleup.common.PaymentDeclinedException;
import com.scaleup.common.PaymentTimeoutException;
import com.scaleup.notification.NotificationSender;
import com.scaleup.payment.PaymentGateway;
import com.scaleup.payment.PaymentOutcome;
import org.springframework.stereotype.Service;

// Decides the order of the stages. Deliberately NOT @Transactional: it calls the short
// transactions in OrderTransactions, and the slow payment and notification calls,
// which must run outside any database transaction.
@Service
public class OrderService {

    private static final String ORDER_CONFIRMATION = "ORDER_CONFIRMATION";

    private final OrderTransactions orderTransactions;
    private final PaymentGateway paymentGateway;
    private final NotificationSender notificationSender;

    public OrderService(OrderTransactions orderTransactions,
                        PaymentGateway paymentGateway,
                        NotificationSender notificationSender) {
        this.orderTransactions = orderTransactions;
        this.paymentGateway = paymentGateway;
        this.notificationSender = notificationSender;
    }

    public OrderResponse placeOrder(OrderRequest request) {
        rejectRepeatedProducts(request);

        // Transaction 1 (short): reserve stock and create the order.
        OrderResponse reserved = orderTransactions.createAndReserve(request);
        Long orderId = reserved.orderId();

        // Stage 4: the slow external call. No database connection is held while it runs.
        PaymentOutcome outcome = paymentGateway.charge(orderId, reserved.totalAmount());

        return switch (outcome) {
            case SUCCESS -> {
                // Transaction 2 (short): record the payment, stock becomes sold, order is PAID.
                orderTransactions.recordPaymentSuccess(orderId, reserved.totalAmount());

                // Stage 6: slow external call, again outside any transaction.
                // Baseline behaviour (P07): if it fails, the order stays PAID and the API returns 500.
                notificationSender.send(orderId, ORDER_CONFIRMATION);

                // Transaction 3 (short): record the notification and confirm the order.
                yield orderTransactions.recordConfirmation(orderId, ORDER_CONFIRMATION);
            }
            case DECLINED -> {
                // The transaction commits first; only then do we report the failure.
                orderTransactions.recordPaymentDeclined(orderId, reserved.totalAmount());
                throw new PaymentDeclinedException(orderId);
            }
            // Baseline behaviour (P12): nothing is released, the order stays STOCK_RESERVED.
            case TIMEOUT -> throw new PaymentTimeoutException(orderId);
            case ERROR -> throw new IllegalStateException(
                    "Unexpected payment error for order " + orderId);
        };
    }

    public OrderResponse getById(Long orderId) {
        return orderTransactions.getById(orderId);
    }

    private void rejectRepeatedProducts(OrderRequest request) {
        long distinctProducts = request.items().stream()
                .map(OrderItemRequest::productId)
                .distinct()
                .count();
        if (distinctProducts != request.items().size()) {
            throw new BadRequestException("The same product appears more than once in the order");
        }
    }
}

package com.scaleup.order;

import com.scaleup.common.NotFoundException;
import com.scaleup.customer.Customer;
import com.scaleup.customer.CustomerService;
import com.scaleup.inventory.InventoryService;
import com.scaleup.notification.Notification;
import com.scaleup.notification.NotificationRepository;
import com.scaleup.notification.NotificationStatus;
import com.scaleup.payment.Payment;
import com.scaleup.payment.PaymentOutcome;
import com.scaleup.payment.PaymentRepository;
import com.scaleup.product.Product;
import com.scaleup.product.ProductService;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.math.BigDecimal;
import java.util.Comparator;
import java.util.List;
import java.util.Map;

// The database transactions of the order flow. Each public method is one transaction.
@Service
public class OrderTransactions {

    private final CustomerService customerService;
    private final ProductService productService;
    private final InventoryService inventoryService;
    private final OrderRepository orderRepository;
    private final PaymentRepository paymentRepository;
    private final NotificationRepository notificationRepository;

    public OrderTransactions(CustomerService customerService,
                             ProductService productService,
                             InventoryService inventoryService,
                             OrderRepository orderRepository,
                             PaymentRepository paymentRepository,
                             NotificationRepository notificationRepository) {
        this.customerService = customerService;
        this.productService = productService;
        this.inventoryService = inventoryService;
        this.orderRepository = orderRepository;
        this.paymentRepository = paymentRepository;
        this.notificationRepository = notificationRepository;
    }

    // Transaction 1: look up, reserve stock, create the order. All or nothing.
    @Transactional
    public OrderResponse createAndReserve(OrderRequest request) {
        Customer customer = customerService.getEntityById(request.customerId());

        List<Long> productIds = request.items().stream()
                .map(OrderItemRequest::productId)
                .toList();
        Map<Long, Product> products = productService.getEntitiesByIds(productIds);

        // Ascending product id order: two orders can never wait for each other in a circle.
        List<OrderItemRequest> sortedItems = request.items().stream()
                .sorted(Comparator.comparing(OrderItemRequest::productId))
                .toList();

        BigDecimal total = BigDecimal.ZERO;
        for (OrderItemRequest item : sortedItems) {
            // Throws InsufficientStockException if there is not enough: everything rolls back.
            inventoryService.reserve(item.productId(), item.quantity());
            Product product = products.get(item.productId());
            total = total.add(product.getPrice().multiply(BigDecimal.valueOf(item.quantity())));
        }

        Order order = new Order(customer, OrderStatus.STOCK_RESERVED, total);
        for (OrderItemRequest item : sortedItems) {
            Product product = products.get(item.productId());
            order.addItem(product, item.quantity(), product.getPrice());
        }

        return OrderResponse.from(orderRepository.save(order));
    }

    // Transaction 2a: payment succeeded. Reserved units become sold, the payment is recorded,
    // the order becomes PAID.
    @Transactional
    public OrderResponse recordPaymentSuccess(Long orderId, BigDecimal amount) {
        Order order = loadOrderWithItems(orderId);
        for (OrderItem item : itemsInProductOrder(order)) {
            inventoryService.markSold(item.getProduct().getId(), item.getQuantity());
        }
        paymentRepository.save(new Payment(orderId, PaymentOutcome.SUCCESS, amount));
        order.markPaid();
        return OrderResponse.from(order);
    }

    // Transaction 2b: payment declined. The stock goes back, the payment is recorded,
    // the order becomes PAYMENT_FAILED.
    @Transactional
    public void recordPaymentDeclined(Long orderId, BigDecimal amount) {
        Order order = loadOrderWithItems(orderId);
        for (OrderItem item : itemsInProductOrder(order)) {
            inventoryService.release(item.getProduct().getId(), item.getQuantity());
        }
        paymentRepository.save(new Payment(orderId, PaymentOutcome.DECLINED, amount));
        order.markPaymentFailed();
    }

    // Transaction 3: the notification was sent. Record it and confirm the order.
    @Transactional
    public OrderResponse recordConfirmation(Long orderId, String notificationType) {
        Order order = orderRepository.findById(orderId)
                .orElseThrow(() -> new NotFoundException("Order " + orderId + " not found"));
        notificationRepository.save(new Notification(orderId, notificationType, NotificationStatus.SENT));
        order.markConfirmed();
        return OrderResponse.from(order);
    }

    @Transactional(readOnly = true)
    public OrderResponse getById(Long orderId) {
        return orderRepository.findById(orderId)
                .map(OrderResponse::from)
                .orElseThrow(() -> new NotFoundException("Order " + orderId + " not found"));
    }

    private Order loadOrderWithItems(Long orderId) {
        return orderRepository.findWithItemsById(orderId)
                .orElseThrow(() -> new NotFoundException("Order " + orderId + " not found"));
    }

    private List<OrderItem> itemsInProductOrder(Order order) {
        return order.getItems().stream()
                .sorted(Comparator.comparing((OrderItem item) -> item.getProduct().getId()))
                .toList();
    }
}

import com.scaleup.simulation.SimulationConfig;
import com.scaleup.simulation.SimulationSettings;
import com.scaleup.simulation.SimulationSupport;
import org.springframework.stereotype.Component;

import java.math.BigDecimal;

@Component
public class SimulatedPaymentGateway implements PaymentGateway {

    private final SimulationSettings settings;
    private final SimulationSupport support;

    public SimulatedPaymentGateway(SimulationSettings settings, SimulationSupport support) {
        this.settings = settings;
        this.support = support;
    }

    @Override
    public PaymentOutcome charge(Long orderId, BigDecimal amount) {
        SimulationConfig config = settings.get();

        // One draw decides the outcome: first slice = timeout, next slice = declined, rest = success.
        double roll = support.nextDouble();

        if (roll < config.paymentTimeoutRate()) {
            support.sleep(config.paymentTimeoutMs());
            return PaymentOutcome.TIMEOUT;
        }

        support.sleep(config.paymentDelayMs());

        if (roll < config.paymentTimeoutRate() + config.paymentDeclineRate()) {
            return PaymentOutcome.DECLINED;
        }
        return PaymentOutcome.SUCCESS;
    }
}

import java.util.Collection;
import java.util.Map;
import java.util.function.Function;
import java.util.stream.Collectors;

@Service
public class ProductService {

    private final ProductRepository productRepository;
    private final InventoryService inventoryService;

    public ProductService(ProductRepository productRepository, InventoryService inventoryService) {
        this.productRepository = productRepository;
        this.inventoryService = inventoryService;
    }

    @Transactional(readOnly = true)
    public ProductResponse getById(Long id) {
        Product product = productRepository.findById(id)
                .orElseThrow(() -> new NotFoundException("Product " + id + " not found"));
        int available = inventoryService.getAvailableQuantity(id);
        return ProductResponse.from(product, available);
    }

    // Loads several products with one query. Every id must exist and the product must be active.
    @Transactional(readOnly = true)
    public Map<Long, Product> getEntitiesByIds(Collection<Long> ids) {
        Map<Long, Product> found = productRepository.findAllById(ids).stream()
                .collect(Collectors.toMap(Product::getId, Function.identity()));
        for (Long id : ids) {
            Product product = found.get(id);
            if (product == null) {
                throw new NotFoundException("Product " + id + " not found");
            }
            if (!product.isActive()) {
                throw new BadRequestException("Product " + id + " is not active");
            }
        }
        return found;
    }
}

            new AtomicReference<>(SimulationConfig.defaults());

    // Increases on every update or reset, so others can tell that the settings were re-applied.
    private final AtomicLong version = new AtomicLong();

    public SimulationConfig get() {
        return current.get();
    }

    public long version() {
        return version.get();
    }

    public void update(SimulationConfig config) {
        current.set(config);
        version.incrementAndGet();
    }

    public void reset() {
        current.set(SimulationConfig.defaults());
        version.incrementAndGet();
    }
}

package com.scaleup.simulation;

import org.springframework.stereotype.Component;

import java.util.Random;

@Component
public class SimulationSupport {

    private final SimulationSettings settings;
    private Random random;
    private long seenVersion;

    public SimulationSupport(SimulationSettings settings) {
        this.settings = settings;
        this.seenVersion = settings.version();
        this.random = new Random(settings.get().randomSeed());
    }

    // A number from 0.0 up to (not including) 1.0.
    // The sequence starts again whenever the settings are updated or reset.
    public synchronized double nextDouble() {
        long currentVersion = settings.version();
        if (currentVersion != seenVersion) {
            seenVersion = currentVersion;
            random = new Random(settings.get().randomSeed());
        }
        return random.nextDouble();
    }

    // Blocks the calling thread, like a real network call that waits for an answer.
    public void sleep(long millis) {
        if (millis <= 0) {
            return;
        }
        try {
            Thread.sleep(millis);
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new IllegalStateException("Simulated call was interrupted", e);
        }
    }
}

                .isInstanceOf(InsufficientStockException.class);

        assertStock(4, 6);
    }

    @Test
    void releaseMovesQuantityBack() {
        inventoryService.reserve(productId, 4);

        inventoryService.release(productId, 4);

        assertStock(10, 0);
    }

    @Test
    void releasingMoreThanReservedFails() {
        assertThatThrownBy(() -> inventoryService.release(productId, 1))
                .isInstanceOf(IllegalStateException.class);

        assertStock(10, 0);
    }

package com.scaleup.notification;

import com.scaleup.simulation.SimulationConfig;
import com.scaleup.simulation.SimulationSettings;
import com.scaleup.simulation.SimulationSupport;
import org.junit.jupiter.api.Test;

import java.util.ArrayList;
import java.util.List;

import static org.assertj.core.api.Assertions.assertThat;
import static org.assertj.core.api.Assertions.assertThatThrownBy;

class SimulatedNotificationSenderTest {

    private final SimulationSettings settings = new SimulationSettings();
    private final SimulatedNotificationSender sender =
            new SimulatedNotificationSender(settings, new SimulationSupport(settings));

    private static SimulationConfig config(long delayMs, double failureRate) {
        return new SimulationConfig(0, 0.0, 0.0, 3000, delayMs, failureRate, 12345);
    }

    // true = the call failed, false = the call worked
    private List<Boolean> sendMany(int times) {
        List<Boolean> failed = new ArrayList<>();
        for (int i = 0; i < times; i++) {
            try {
                sender.send(1L, "ORDER_CONFIRMATION");
                failed.add(false);
            } catch (NotificationFailedException e) {
                failed.add(true);
            }
        }
        return failed;
    }

    @Test
    void defaultSettingsNeverFail() {
        assertThat(sendMany(100)).containsOnly(false);
    }

    @Test
    void delayIsApplied() {
        settings.update(config(200, 0.0));

        long start = System.nanoTime();
        sender.send(1L, "ORDER_CONFIRMATION");
        long elapsedMs = (System.nanoTime() - start) / 1_000_000;

        // A little below 200 is allowed because timers are not perfectly exact.
        assertThat(elapsedMs).isGreaterThanOrEqualTo(190);
    }

    @Test
    void failureRateOfOneAlwaysFails() {
        settings.update(config(0, 1.0));

        assertThatThrownBy(() -> sender.send(1L, "ORDER_CONFIRMATION"))
                .isInstanceOf(NotificationFailedException.class)
                .hasMessageContaining("order 1");
    }

    @Test
    void aFailingCallStillTakesTheDelay() {
        settings.update(config(200, 1.0));

        long start = System.nanoTime();
        assertThatThrownBy(() -> sender.send(1L, "ORDER_CONFIRMATION"))
                .isInstanceOf(NotificationFailedException.class);
        long elapsedMs = (System.nanoTime() - start) / 1_000_000;

        assertThat(elapsedMs).isGreaterThanOrEqualTo(190);
    }

    @Test
    void sameSettingsGiveTheSameSequence() {
        settings.update(config(0, 0.5));
        List<Boolean> first = sendMany(20);

        // Applying the settings again restarts the random sequence.
        settings.update(config(0, 0.5));
        List<Boolean> second = sendMany(20);

        assertThat(second).isEqualTo(first);
    }

    @Test
    void observedFailureShareIsCloseToTheConfiguredRate() {
        settings.update(config(0, 0.3));

        long failures = sendMany(10_000).stream().filter(failed -> failed).count();

        assertThat(failures).isBetween(2_700L, 3_300L);
    }
}

package com.scaleup.payment;

import com.scaleup.simulation.SimulationConfig;
import com.scaleup.simulation.SimulationSettings;
import com.scaleup.simulation.SimulationSupport;
import org.junit.jupiter.api.Test;

import java.math.BigDecimal;
import java.util.ArrayList;
import java.util.List;

import static org.assertj.core.api.Assertions.assertThat;

class SimulatedPaymentGatewayTest {

    private final SimulationSettings settings = new SimulationSettings();
    private final SimulatedPaymentGateway gateway =
            new SimulatedPaymentGateway(settings, new SimulationSupport(settings));

    private static SimulationConfig config(long delayMs, double declineRate, double timeoutRate, long timeoutMs) {
        return new SimulationConfig(delayMs, declineRate, timeoutRate, timeoutMs, 0, 0.0, 12345);
    }

    private List<PaymentOutcome> charge(int times) {
        List<PaymentOutcome> outcomes = new ArrayList<>();
        for (int i = 0; i < times; i++) {
            outcomes.add(gateway.charge(1L, BigDecimal.TEN));
        }
        return outcomes;
    }

    @Test
    void defaultSettingsAlwaysSucceed() {
        assertThat(charge(100)).containsOnly(PaymentOutcome.SUCCESS);
    }

    @Test
    void delayIsApplied() {
        settings.update(config(200, 0.0, 0.0, 3000));

        long start = System.nanoTime();
        PaymentOutcome outcome = gateway.charge(1L, BigDecimal.TEN);
        long elapsedMs = (System.nanoTime() - start) / 1_000_000;

        assertThat(outcome).isEqualTo(PaymentOutcome.SUCCESS);
        // A little below 200 is allowed because timers are not perfectly exact.
        assertThat(elapsedMs).isGreaterThanOrEqualTo(190);
    }

    @Test
    void declineRateOfOneAlwaysDeclines() {
        settings.update(config(0, 1.0, 0.0, 3000));

        assertThat(charge(20)).containsOnly(PaymentOutcome.DECLINED);
    }

    @Test
    void timeoutRateOfOneBlocksThenTimesOut() {
        settings.update(config(0, 0.0, 1.0, 300));

        long start = System.nanoTime();
        PaymentOutcome outcome = gateway.charge(1L, BigDecimal.TEN);
        long elapsedMs = (System.nanoTime() - start) / 1_000_000;

        assertThat(outcome).isEqualTo(PaymentOutcome.TIMEOUT);
        assertThat(elapsedMs).isGreaterThanOrEqualTo(290);
    }

    @Test
    void sameSettingsGiveTheSameSequence() {
        settings.update(config(0, 0.5, 0.0, 3000));
        List<PaymentOutcome> first = charge(20);

        // Applying the settings again restarts the random sequence.
        settings.update(config(0, 0.5, 0.0, 3000));
        List<PaymentOutcome> second = charge(20);

        assertThat(second).isEqualTo(first);
    }

    @Test
    void observedDeclineShareIsCloseToTheConfiguredRate() {
        settings.update(config(0, 0.3, 0.0, 3000));

        long declined = charge(10_000).stream()
                .filter(outcome -> outcome == PaymentOutcome.DECLINED)
                .count();

        assertThat(declined).isBetween(2_700L, 3_300L);
    }
}
